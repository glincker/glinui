  ## Usage

  Pass an array of `items` with `id`, `label`, and optional `href`. The last item is highlighted as the current page.

  ## Examples

  ### Variants

  `GlassBreadcrumb` is glass by default, with a readable opacity floor. It also accepts the shared vocabulary: `glinr` raises the current page as a pill, `plain` is flat and shadcn-like. Glass looks frosted only over a colourful or photographic backdrop: pick one in the stage header.

  <ExampleBlock
    code={`import { GlassBreadcrumb } from "@glinui/ui"

const variants = ["glinr", "plain", "solid", "soft", "outline", "ghost", "gradient", "glass"] as const

export function GlassBreadcrumbVariantsDemo() {
  const items = [
    { id: "home", label: "Home", href: "/" },
    { id: "docs", label: "Docs", href: "/docs" },
    { id: "breadcrumb", label: "Breadcrumb" }
  ]
  return (
    <div className="flex flex-col items-start gap-3">
      {variants.map((variant) => (
        <GlassBreadcrumb key={variant} variant={variant} items={items} />
      ))}
    </div>
  )
}`}
  >
    <div className="flex flex-col items-start gap-3">
      {["glinr", "plain", "solid", "soft", "outline", "ghost", "gradient", "glass"].map((variant) => (
        <GlassBreadcrumb
          key={variant}
          variant={variant}
          items={[
            { id: "home", label: "Home", href: "/" },
            { id: "docs", label: "Docs", href: "/docs" },
            { id: "breadcrumb", label: "Breadcrumb" }
          ]}
        />
      ))}
    </div>
  </ExampleBlock>

  ### Collapsed

  Use `maxItems` to collapse middle items into an ellipsis.

  <ExampleBlock
    code={`import { GlassBreadcrumb } from "@glinui/ui"

export function GlassBreadcrumbCollapsedDemo() {
  return (
    <div className="flex items-center justify-center p-8">
      <GlassBreadcrumb
        maxItems={3}
        items={[
          { id: "home", label: "Home", href: "/" },

          { id: "signature", label: "Signature", href: "/docs/components" },

  ## Accessibility

  - Renders as `<nav aria-label="Breadcrumb">` with an `<ol>` list.
  - The current page has `aria-current="page"`.
  - Separators are `aria-hidden`.

  ## Reduced Motion

  No animations, the component is fully static.
